bool DatabaseManager::initDatabase()
{
    db = QSqlDatabase::addDatabase("QSQLITE");
    db.setDatabaseName(SQL_NAME);

    if(!db.open())
    {
        qWarning() << "数据库打开失败" << db.lastError().text();
        return false;
    }

    QSqlQuery query;
    bool success = query.exec("CREATE TABLE IF NOT EXISTS information("
                              "id INTEGER PRIMARY KEY,"
                              "name TEXT,"
                              "age INTEGER,"
                              "sex TEXT,"
                              "class TEXT,"
                              "email TEXT,"
                              "finger_id INTEGER UNIQUE"
                              ")");
    if(!success)
    {
        qWarning() << "建表失败" << query.lastError().text();
        return false;
    }

    return true;

}

bool DatabaseManager::addInformation(const quint32 &id, const QString &name, const quint8 &age, const QString &sex, const QString &schoolClass, const QString &email, const quint16 &finger_id)
{
    QSqlQuery query;
    query.prepare("INSERT INTO information "
                  "(id,name,age,sex,class,email,finger_id) "
                  "VALUES(:id,:name,:age,:sex,:class,:email,:finger_id)");

    query.bindValue(":id",id);
    query.bindValue(":name",name);
    query.bindValue(":age",age);
    query.bindValue(":sex",sex);
    query.bindValue(":class",schoolClass);
    query.bindValue(":email",email);
    query.bindValue(":finger_id",finger_id);

    bool success = query.exec();

    if(!success)
    {
        qWarning() << "录入学生信息失败：" << query.lastError().text();
        return false;
    }

    return true;
}

bool DatabaseManager::addInformation(const quint16 &finger_id)
{
    QSqlQuery query;

    query.prepare(
        "INSERT INTO information (finger_id) "
        "VALUES (:finger_id)"
        );

    query.bindValue(":finger_id", finger_id);

    if (!query.exec())
    {
        qWarning() << "创建学生信息失败：" << query.lastError().text();
        return false;
    }

    return true;
}

QVariantMap DatabaseManager::getInfoByFinger(quint16 finger_id)
{
    QSqlQuery query;
    QVariantMap info;
    query.prepare("SELECT id,name,age,sex,class,email "
                  "FROM information "
                  "WHERE finger_id = :finger_id");
    query.bindValue(":finger_id",finger_id);

    bool success = query.exec();
    if(!success)
    {
        qWarning() << "查询失败" << query.lastError().text();
        return info;
    }
    success &= query.next();
    if(!success)
    {
        qWarning() << "查询无此人" << query.lastError().text();
        return info;
    }

    info["id"] = query.value("id").toUInt();
    info["name"] = query.value("name").toString();
    info["age"] = query.value("age").toUInt();
    info["sex"] = query.value("sex").toString();
    info["class"] = query.value("class").toString();
    info["email"] = query.value("email").toString();

    return info;
}

bool DatabaseManager::isInfoExistsByFinger(quint16 finger_id)
{
    QSqlQuery query;
    query.prepare("SELECT EXISTS("
                                 "SELECT 1 "
                                 "FROM information "
                                 "WHERE finger_id = :finger_id);");
    query.bindValue(":finger_id",finger_id);
    bool success = query.exec();
    if(!success)
    {
        qWarning() << "查询失败" << query.lastError().text();
        return false;
    }

    success = query.next();

    if(!success)
    {
        qWarning() << "查询结果为空" << query.lastError().text();
    }

    bool exists = query.value(0).toBool();

    return exists;

}

void DatabaseManager::verifyFingerInfo(const QByteArray &data)
{
    quint16 cnt = (data[0]<<8) + data[1];
    for(quint16 i = 1;i<=cnt;i++)
    {
        if(!isInfoExistsByFinger(i))
        {
            addInformation(i);
        }
    }
    emit updateTableView();
}

void DatabasePanel::showTableContextMenu(const QPoint &pos)
{
    QModelIndex index = tableView->indexAt(pos);

    if (!index.isValid())
        return;

    tableView->setCurrentIndex(index);

    QMenu menu(this);

    QAction *deleteAction = menu.addAction("删除");

    connect(deleteAction, &QAction::triggered,
            this, &DatabasePanel::deleteSelectedRow);

    menu.exec(tableView->viewport()->mapToGlobal(pos));
}

void DatabasePanel::deleteSelectedRow()
{
    QModelIndex index = tableView->currentIndex();



    if (!index.isValid())
        return;

    emit deleteInfoByIndex(getFingerIdByRow(static_cast<int>(index.row())));

    int row = index.row();

    if (!tableModel->removeRow(row))
    {
        return;
    }


    if (!tableModel->submitAll())
    {

        tableModel->revertAll();
        return;
    }


    tableModel->select();
}

quint16 DatabasePanel::getFingerIdByRow(int row)
{
    
    if (row < 0 || row >= tableModel->rowCount()) {
        return 0; 
    }

    
    QSqlRecord record = tableModel->record(row);

    
    return static_cast<quint16>(record.value("finger_id").toUInt());
}

void DatabasePanel::updateTableView()
{
    tableModel->select();
}

void DeviceControl::parseCmd(quint8 cmd,const QByteArray &data)
{
    if(cmd>=0xA0 && cmd<=0xAF)
    {
        switch(cmd)
        {
        case ENROLL_MODE:
        {
            emit enrollMode(data);

            break;
        }
        case IDLE_MODE:
        {
            emit idleMode(data);
            break;
        }
        case UPDATE_FINGER_CNT:
        {
            emit updateFingerCount(data);
            break;
        }
        case DELETE_FINGER_SUCCEED:
        {
            qDebug() << "下位机删除指纹成功";
            break;
        }
        case DELETE_FINGER_FAILED:
        {
            qDebug() << "下位机删除指纹失败";
            break;
        }
        }
    }
    else if(cmd>=0xC0&&cmd<=0xDF)
    {
        parseEnrollCmd(cmd,data);
    }
    else if(cmd>=0xE0&&cmd<0xEF)
    {
        parseVerifyCmd(cmd,data);
    }
    else
    {
        qWarning() << "Unknown command received:" << Qt::hex << cmd;
    }
}

void DeviceControl::deleteFingerByIndex(quint16 cnt)
{
    QByteArray tmp;
    tmp.resize(2);
    tmp[0] = static_cast<char>((cnt >> 8) & 0xFF);
    tmp[1] = static_cast<char>(cnt & 0xFF);
    QByteArray data = protocol->Builder(DELETE_FINGER,tmp);
    tcpClient->send(data);
}



void DeviceControl::toggleWorkMode()
{
    QByteArray data =  protocol->Builder(TOGGLE_MODE,"");
    tcpClient->send(data);
}

void DeviceControl::onConnectedReceived(QList<QByteArray> &data)
{
    for(QByteArray i:data)
    {
        
        quint8 cmd = static_cast<quint8>(i[0]);
        i.remove(0,1);

        
        parseCmd(cmd,i);
    }
}

QByteArray Protocol::Builder(quint8 cmd, const QByteArray &data)
{
    QByteArray frame;
    frame.append((char)FRAME_HEADER_1);
    frame.append((char)FRAME_HEADER_2);
    frame.append((char)cmd);
    frame.append((char)data.size());

    if(!data.isEmpty()){
        frame.append(data);
    }

    frame.append((char)FRAME_TAIL_1);
    frame.append((char)FRAME_TAIL_2);

    return frame;
}


QList<QByteArray> Protocol::Parser(const QByteArray &byte)
{
    m_buffer.append(byte);
    QList<QByteArray> result;

    
    QByteArray headerSeq;
    headerSeq.append((char)FRAME_HEADER_1);
    headerSeq.append((char)FRAME_HEADER_2);

    while(1)
    {
        
        int headIdx = m_buffer.indexOf(headerSeq);
        if(headIdx == -1)
        {
            
            if(m_buffer.endsWith((char)FRAME_HEADER_1)) {
                m_buffer = m_buffer.right(1);
            } else {
                m_buffer.clear();
            }
            break;
        }

        
        if(headIdx > 0)
        {
            m_buffer.remove(0, headIdx);
        }

        
        if(m_buffer.size() < 6)
        {
            break;
        }

        
        quint8 len = (quint8)m_buffer[3];
        int frameTotalSize = 6 + len;

        
        if(m_buffer.size() < frameTotalSize)
        {
            break;
        }

        
        if((quint8)m_buffer[frameTotalSize - 2] == FRAME_TAIL_1 &&
            (quint8)m_buffer[frameTotalSize - 1] == FRAME_TAIL_2)
        {
            
            quint8 cmd = (quint8)m_buffer[2];
            QByteArray data = m_buffer.mid(4, len);

            
            QByteArray packet;
            packet.append((char)cmd);
            packet.append(data);
            result.append(packet);

            
            m_buffer.remove(0, frameTotalSize);
        }
        else
        {
            
            m_buffer.remove(0, 2);
        }
    }
    return result;
}

TcpConnection* TcpClient::connectToHost(const QString &address, quint16 port)
{
    TcpConnection* connection = new TcpConnection(this);
    QObject::connect(connection,&TcpConnection::received,this,&TcpClient::onConnectionReceived);

    
    QObject::connect(connection, &TcpConnection::connected, this, [=](){
        qDebug() << "网络底层真正连接成功！地址:" << address << "端口:" << port;
        QMessageBox::information(nullptr, "提示", "设备连接成功！");
    });

    
    QObject::connect(connection, &TcpConnection::errorOccurred, this, [=](QAbstractSocket::SocketError err){
        qDebug() << "连接失败，错误码:" << err;
        QMessageBox::warning(nullptr, "警告", "无法连接到设备，请检查IP和端口！");
    });

    connection->connectToHost(address,port);
    this->tcpConnection = connection;
    return connection;
}

    :QObject(parent)
{
    this->protocol = new Protocol(this);
    this->Server = new QTcpServer(this);
    QObject::connect(Server,&QTcpServer::newConnection,this,&TcpServer::onNewConnection);

void TcpServer::onNewConnection()
{
    QTcpSocket* socket = Server->nextPendingConnection();
    TcpConnection* connection = new TcpConnection(socket,this);
    this->TcpList.append(connection);

    QObject::connect(connection,&TcpConnection::disconnected,this,&TcpServer::onDisconnected);
}

void TcpServer::onDisconnected()
{
    TcpConnection* connection = qobject_cast<TcpConnection*>(sender());
    if(connection==nullptr)
    {
        return;
    }
    TcpList.removeOne(connection);

    connection->deleteLater();
}
